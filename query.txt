-- Insert staff-commission.export permission
INSERT INTO permissions (module_id, name, slug, created_at, updated_at)
SELECT id, 'Export Staff Commission Invoices', 'staff-commission.export', NOW(), NOW()
FROM modules 
WHERE slug = 'staff-commission'
ON DUPLICATE KEY UPDATE name = 'Export Staff Commission Invoices', updated_at = NOW();

-- Map staff-commission.export permission to admin role (if not already mapped)
INSERT INTO role_permissions (role_id, permission_id, allowed, created_at, updated_at)
SELECT admin_role.id, perm.id, 1, NOW(), NOW()
FROM roles admin_role
CROSS JOIN permissions perm
WHERE admin_role.slug = 'admin'
  AND perm.slug = 'staff-commission.export'
  AND NOT EXISTS (
      SELECT 1 FROM role_permissions rp 
      WHERE rp.role_id = admin_role.id 
        AND rp.permission_id = perm.id
  );
