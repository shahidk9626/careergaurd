-- ==========================================
-- 2. COMMISSION PAYMENT SYSTEM
-- ==========================================

-- Add commission amount column to plans table
ALTER TABLE plans ADD COLUMN commission_amount DECIMAL(10, 2) NOT NULL DEFAULT 0.00 AFTER compensation_amount;

-- Create staff commission payments batch table
CREATE TABLE staff_commission_payments (
    id BIGINT UNSIGNED AUTO_INCREMENT PRIMARY KEY,
    staff_id BIGINT UNSIGNED NOT NULL,
    batch_reference VARCHAR(255) NOT NULL UNIQUE,
    total_policies INT NOT NULL,
    total_commission_amount DECIMAL(10, 2) NOT NULL,
    payment_proof VARCHAR(255) NULL,
    description TEXT NULL,
    payment_date DATE NOT NULL,
    created_by BIGINT UNSIGNED NOT NULL,
    created_at TIMESTAMP NULL,
    updated_at TIMESTAMP NULL,
    FOREIGN KEY (staff_id) REFERENCES users(id) ON DELETE CASCADE,
    FOREIGN KEY (created_by) REFERENCES users(id) ON DELETE CASCADE
);

-- Create staff commission payment details status table
CREATE TABLE staff_commission_payment_details (
    id BIGINT UNSIGNED AUTO_INCREMENT PRIMARY KEY,
    payment_id BIGINT UNSIGNED NULL,
    purchased_plan_id BIGINT UNSIGNED NOT NULL,
    customer_id BIGINT UNSIGNED NOT NULL,
    plan_id BIGINT UNSIGNED NOT NULL,
    commission_amount DECIMAL(10, 2) NOT NULL,
    status VARCHAR(255) NOT NULL DEFAULT 'Pending', -- Pending, Paid, Hold, Rejected
    description TEXT NULL, -- Hold or rejection reason
    created_at TIMESTAMP NULL,
    updated_at TIMESTAMP NULL,
    FOREIGN KEY (payment_id) REFERENCES staff_commission_payments(id) ON DELETE CASCADE,
    FOREIGN KEY (purchased_plan_id) REFERENCES purchased_plans(id) ON DELETE CASCADE,
    FOREIGN KEY (customer_id) REFERENCES users(id) ON DELETE CASCADE,
    FOREIGN KEY (plan_id) REFERENCES plans(id) ON DELETE CASCADE
);
